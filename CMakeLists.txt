cmake_minimum_required(VERSION 3.24)

# The deployment target must be known before project() so CMake's own -mmacosx-version-min agrees with --target below.
if(APPLE OR CMAKE_HOST_APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version")
endif()

project(Zoosters LANGUAGES C CXX)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(ZoostersOptions)
include(ZoostersDependencies)
include(ZoostersFlags)

set(ZOOSTERS_SRC "${CMAKE_CURRENT_SOURCE_DIR}/src")
set(GAME_DIR "${ZOOSTERS_SRC}/game")
set(DATA_DIR "${ZOOSTERS_SRC}/data")
set(PORT_DIR "${ZOOSTERS_SRC}/port")
set(DEV_DIR "${ZOOSTERS_SRC}/dev")

# ---------------------------------------------------------------------------------------------------------------------------
# Sources. The order is the link order: static initialisers run in it, so it is kept exactly as listed (game, data, port in the
# order below, fixes, options, debug UI, ImGui, librw, glad).

# src/game: every source; with the SDL window, PumpMessages comes from the port's SDL backend (window_sdl.cpp) instead.
file(GLOB_RECURSE game_sources CONFIGURE_DEPENDS "${GAME_DIR}/*.cpp")
list(SORT game_sources)
if(ZOOSTERS_WINDOW STREQUAL "sdl")
  list(REMOVE_ITEM game_sources "${GAME_DIR}/framework/startup/win32/pumpmessages.cpp")
endif()

file(GLOB_RECURSE data_sources CONFIGURE_DEPENDS "${DATA_DIR}/*.cpp")
list(SORT data_sources)

set(port_sources
  port/shim/rwshim port/shim/rwcore port/shim/rwrender port/shim/rwptank port/shim/rwviewcheck port/shim/rt2d port/shim/rwaudio
  port/shim/rwaudio_wavedict port/shim/rwaudio_al
  port/provisional/plugins port/provisional/thunks port/provisional/game_stubs port/provisional/game_access
  port/provisional/userdir port/provisional/rw_unimplemented port/provisional/script_stubs port/provisional/mouse port/shim/dinput_port)
if(ZOOSTERS_BINK)
  list(APPEND port_sources port/bink/bink port/bink/binkdec)   # binkdec: LGPL-2.1-or-later (NOTICE.md)
else()
  list(APPEND port_sources port/shim/binkshim)
endif()
if(ZOOSTERS_DEV_HARNESS)
  list(APPEND port_sources dev/gaddr dev/gtrace_port)
else()
  list(APPEND port_sources port/harness_off)
endif()
list(APPEND port_sources port/loader/port_run port/runtime)
if(NOT ZOOSTERS_OS STREQUAL "sdl")
  list(APPEND port_sources port/fault64)          # Windows crash logging of runtime.cpp's fault handler (x64)
endif()
list(APPEND port_sources port/provisional/game_traps)
if(ZOOSTERS_DEV_HARNESS)
  list(APPEND port_sources dev/semtrace)
endif()
if(ZOOSTERS_OS STREQUAL "sdl")
  list(APPEND port_sources port/loader/player_launch)
endif()
list(APPEND port_sources port/os/os_${ZOOSTERS_OS})
if(ZOOSTERS_CALLTRACE)
  list(APPEND port_sources dev/calltrace)
endif()
list(APPEND port_sources port/os/settings_${ZOOSTERS_SETTINGS})
if(ZOOSTERS_PLATFORM STREQUAL "null")
  list(APPEND port_sources port/shim/win32shim port/provisional/window)
else()
  if(ZOOSTERS_WINDOW STREQUAL "sdl")
    list(APPEND port_sources port/provisional/window_sdl)
  else()
    list(APPEND port_sources port/provisional/window_w32)
  endif()
  if(ZOOSTERS_PLATFORM STREQUAL "d3d9" AND ZOOSTERS_DEV_HARNESS)
    list(APPEND port_sources dev/d3dcap)
  endif()
endif()
if(ZOOSTERS_WINDOW STREQUAL "sdl" AND NOT ZOOSTERS_PLATFORM STREQUAL "null")
  list(APPEND port_sources port/provisional/gamepad_sdl)
else()
  list(APPEND port_sources port/provisional/gamepad_none)
endif()
if(ZOOSTERS_TIER_INDEX GREATER_EQUAL 1)
  list(APPEND port_sources port/fixes/display)
endif()
list(TRANSFORM port_sources PREPEND "${ZOOSTERS_SRC}/")
list(TRANSFORM port_sources APPEND ".cpp")
if(ZOOSTERS_TIER_INDEX GREATER_EQUAL 2)
  list(APPEND port_sources "${PORT_DIR}/fixes/widescreen.cpp" "${PORT_DIR}/fixes/wide_scenery.cpp")
  file(GLOB_RECURSE options_sources CONFIGURE_DEPENDS "${PORT_DIR}/options/*.cpp")
  list(SORT options_sources)
  list(APPEND port_sources ${options_sources})
endif()
if(ZOOSTERS_DEBUGUI)
  foreach(s rw panels physics triggers practice freecam studio camtour vars)
    list(APPEND port_sources "${PORT_DIR}/debugui/debugui_${s}.cpp")
  endforeach()
endif()

set(librw_sources)
foreach(d src src/ps2 src/d3d src/gl src/lodepng)
  file(GLOB s CONFIGURE_DEPENDS "${ZOOSTERS_LIBRW_DIR}/${d}/*.cpp")
  list(SORT s)
  list(APPEND librw_sources ${s})
endforeach()
if(NOT librw_sources)
  message(FATAL_ERROR "librw: no sources in ${ZOOSTERS_LIBRW_DIR} (set ZOOSTERS_LIBRW_DIR or fetch vendor/librw)")
endif()

# ---------------------------------------------------------------------------------------------------------------------------
# Object libraries: one per flag set (the game and data code, the port + librw, ImGui).

add_library(zoosters_game OBJECT ${game_sources})
add_library(zoosters_data OBJECT ${data_sources})
foreach(t zoosters_game zoosters_data)
  zoosters_game_flags(${t})
endforeach()
if(ZOOSTERS_CALLTRACE)
  zoosters_calltrace_flags(zoosters_game "${GAME_DIR}")
endif()

add_library(zoosters_port OBJECT ${port_sources})
zoosters_port_flags(zoosters_port)

add_library(zoosters_librw OBJECT ${librw_sources})
zoosters_port_flags(zoosters_librw)

set(objects zoosters_game zoosters_data zoosters_port)
if(ZOOSTERS_DEBUGUI)
  set(imgui_sources)
  foreach(s imgui imgui_draw imgui_widgets imgui_tables ImGuizmo)
    list(APPEND imgui_sources "${ZOOSTERS_IMGUI_DIR}/${s}.cpp")
  endforeach()
  add_library(zoosters_imgui OBJECT ${imgui_sources})
  zoosters_base_flags(zoosters_imgui)
  target_include_directories(zoosters_imgui PRIVATE "${ZOOSTERS_IMGUI_DIR}")
  list(APPEND objects zoosters_imgui)
endif()
list(APPEND objects zoosters_librw)
if(ZOOSTERS_PLATFORM STREQUAL "gl3")
  add_library(zoosters_glad OBJECT "${ZOOSTERS_LIBRW_DIR}/src/gl/glad/glad.c")
  zoosters_port_flags(zoosters_glad)
  list(APPEND objects zoosters_glad)
endif()

# ---------------------------------------------------------------------------------------------------------------------------
add_executable(zoosters)
foreach(o ${objects})
  target_sources(zoosters PRIVATE $<TARGET_OBJECTS:${o}>)
endforeach()
if(ZOOSTERS_CALLTRACE AND ZOOSTERS_MSVC AND ZOOSTERS_ARCH STREQUAL "x64")
  enable_language(ASM_MASM)                       # the x64 _penter / _pexit hooks (no inline asm on x64)
  target_sources(zoosters PRIVATE "${DEV_DIR}/calltrace_x64.asm")
endif()
if(WIN32)
  enable_language(RC)
  target_sources(zoosters PRIVATE res/zoosters.rc)   # the executable's icon
endif()
zoosters_link(zoosters)

include(ZoostersRuntime)
zoosters_runtime_files(zoosters)
